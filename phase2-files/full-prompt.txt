const LEGACY_ERROR_MAP = Object.freeze({
    NOT_FOUND: 'ENOENT',
    NOT_A_DIRECTORY: 'ENOTDIR',
    NAME_CONFLICT: 'EEXIST',
    INVALID_NAME: 'EINVAL',
    INVALID_PATH: 'EINVAL',

function createOsError(code, message, detail) {
    const Ctor = window.OsError || Error;
    const err = Ctor === Error ? new Error(message || code) : new Ctor(code, message, detail);
    err.name = 'OsError';
    err.code = code;
    if (detail !== undefined) err.detail = detail;
    return err;
}

function createFsError(code, message, detail) {
    const Ctor = window.FsError || window.OsError || Error;
    const err = Ctor === Error ? new Error(message || code) : new Ctor(code, message, detail);
    err.name = 'FsError';
    err.code = code;
    if (detail !== undefined) err.detail = detail;
    return err;
}

function normalizeFsError(err) {
    let code = 'EINVAL';
    let message = 'Unknown error';
    let detail;
    let name = 'FsError';

    if (typeof err === 'string') {
        message = err;
    } else if (err && typeof err === 'object') {
        code = err.code || code;
        message = err.message || message;
        detail = err.detail;
        name = err.name === 'OsError' ? 'OsError' : 'FsError';
    }

    if (Object.prototype.hasOwnProperty.call(LEGACY_ERROR_MAP, code)) {
        code = LEGACY_ERROR_MAP[code];
    }

    const Ctor = name === 'OsError'
        ? (window.OsError || Error)
        : (window.FsError || window.OsError || Error);

    const normalized = Ctor === Error ? new Error(message) : new Ctor(code, message, detail);
    normalized.name = name;
    normalized.code = code;
    if (detail !== undefined) normalized.detail = detail;
    return normalized;
}

function formatError(err) {
    const normalized = normalizeFsError(err);
    const code = normalized.code || 'ERROR';
    const message = normalized.message || 'Unknown error';
    let extra = '';

    if (normalized.detail && typeof normalized.detail === 'object') {
        const p = normalized.detail.path || normalized.detail.resource || normalized.detail.parent;
        if (p) extra = ` (${p})`;
    }

    return `${code}: ${message}${extra}`;
}

function getSyscallFn() {
    if (window.KernelSyscalls && typeof window.KernelSyscalls.syscall === 'function') {
        return window.KernelSyscalls.syscall.bind(window.KernelSyscalls);
    }

    if (window.Kernel && typeof window.Kernel.syscall === 'function') {
        return window.Kernel.syscall.bind(window.Kernel);
    }

    return null;
}

let vfsRoot = null;
let rootDetectionPromise = null;

async function syscallFs(name, args) {
    const call = getSyscallFn();

    if (!call) {
        throw createOsError('EBUSY', 'Filesystem service unavailable', { syscall: name });
    }

    let result;

    try {
        result = await call(getTerminalContext(), name, args);
    } catch (err) {
        throw normalizeFsError(err);
    }

    if (result && typeof result === 'object' && Object.prototype.hasOwnProperty.call(result, 'ok')) {
        if (!result.ok) {
            throw normalizeFsError(result.error || result.message || 'Syscall failed');
        }

        return result.value;
    }

    return result;
}

async function statRaw(fsPath) {
    return syscallFs('fs.stat', { path: fsPath });
}

async function detectVfsRoot(force = false) {
    if (vfsRoot && !force) return vfsRoot;
    if (rootDetectionPromise && !force) return rootDetectionPromise;

    rootDetectionPromise = (async () => {
        const candidates = ['/', 'C:\\'];

        for (const candidate of candidates) {
            try {
                await statRaw(candidate);
                vfsRoot = candidate;
                return vfsRoot;
            } catch (err) {
                const code = err && err.code;

                if (code === 'ENOENT' || code === 'NOT_FOUND') {
                    continue;
                }

                /*
                 * If a root exists but denies stat, treat it as the root
                 * instead of falling back to a false path.
                 */
                if (code === 'EACCES' || code === 'EPERM' || code === 'PROFILE_SCOPE') {
                    vfsRoot = candidate;
                    return vfsRoot;
                }

                if (code === 'EBUSY') {
                    vfsRoot = '/';
                    return vfsRoot;
                }
            }
        }

        vfsRoot = '/';
        return vfsRoot;
    })();

    try {
        return await rootDetectionPromise;
    } finally {
        rootDetectionPromise = null;
    }
}

function toFsPath(absPath) {
    const path = String(absPath || '/');

    if (/^[A-Za-z]:\\/.test(path)) {
        return path;
    }

    if (!vfsRoot || vfsRoot === '/') {
        return path;
    }

    if (vfsRoot === 'C:\\') {
        if (path === '/') return 'C:\\';
        return 'C:\\' + path.split('/').filter(Boolean).join('\\');
    }

    const root = String(vfsRoot).replace(/\/+$/, '');
    return path === '/' ? root : root + path;
}

async function statPath(absPath) {
    await detectVfsRoot();
    return statRaw(toFsPath(absPath || '/'));
}

function isMissingError(err) {
    const code = err && err.code;
    return code === 'ENOENT' || code === 'NOT_FOUND';
}

async function tryStatPath(absPath) {
    try {
        return await statPath(absPath);
    } catch (err) {
        const normalized = normalizeFsError(err);
        if (isMissingError(normalized)) return null;
        throw normalized;
    }
}

function isDirectory(node) {
    return !!(node && (node.type === 'directory' || node.type === 'dir'));
}

function isFile(node) {
    return !!(node && node.type === 'file');
}

async function listPath(absPath) {
    await detectVfsRoot();

    const value = await syscallFs('fs.list', { path: toFsPath(absPath || '/') });
    let entries = [];

    if (Array.isArray(value)) {
        entries = value;
    } else if (value && Array.isArray(value.entries)) {
        entries = value.entries;
    } else if (value && typeof value === 'object') {
        entries = Object.keys(value).map((key) => {
            const item = value[key];

            if (item && typeof item === 'object') {
                return Object.assign({ name: key }, item);
            }

            return { name: key, type: item };
        });
    }

    return entries
        .filter((entry) => entry && (entry.name || entry.id))
        .map((entry) => Object.assign({}, entry, { name: entry.name || String(entry.id) }));
}

async function readPath(absPath) {
    await detectVfsRoot();

    const value = await syscallFs('fs.read', { path: toFsPath(absPath || '/') });

    if (typeof value === 'string') return value;
    if (value && typeof value.content === 'string') return value.content;
    if (value && typeof value.data === 'string') return value.data;

    return '';
}

async function writePath(absPath, content) {
    await detectVfsRoot();

    const safeContent = content === null || content === undefined ? '' : String(content);

    await syscallFs('fs.write', {
        path: toFsPath(absPath || '/'),
        content: safeContent
    });
}

async function removePath(absPath) {
    await detectVfsRoot();
    await syscallFs('fs.remove', { path: toFsPath(absPath || '/') });
}

function parentPath(absPath) {
    if (!absPath || absPath === '/') return '/';

    const segments = String(absPath).split('/').filter(Boolean);
    segments.pop();

    return segments.length ? `/${segments.join('/')}` : '/';
}

function pathBasename(absPath) {
    const segments = String(absPath || '').split('/').filter(Boolean);
    return segments.pop() || '/';
}

function joinPath(dir, name) {
    const cleanDir = String(dir || '/');
    const cleanName = String(name || '').replace(/^\/+|\/+$/g, '');

    if (!cleanName) return cleanDir;
    if (cleanDir === '/') return `/${cleanName}`;

    return `${cleanDir}/${cleanName}`;
}

    return !vfsRoot || vfsRoot === '/';

    if (!name) {
        throw createFsError('EINVAL', 'Missing name');

    try {
        const key = 'terminal_saved_files_v1';
        const raw = localStorage.getItem(key);
        const map = raw ? JSON.parse(raw) : {};

            content: String(content || ''),

            throw createFsError('ENOSPC', 'Browser storage is full', { name });

        throw normalizeFsError(err);

async function getCompletions(input) {
    const raw = String(input || '');
    const trailingSpace = /\s$/.test(raw);
    const trimmed = raw.replace(/\s+$/, '');
    const tokens = trimmed ? trimmed.split(/\s+/) : [];
    const commandPosition = tokens.length <= 1 && !trailingSpace;
    const partial = commandPosition
        ? (tokens[0] || '')
        : (trailingSpace ? '' : tokens[tokens.length - 1]);

    let candidates = builtInCommands.slice();

    try {
        const entries = await listPath(cwd);
        candidates = candidates.concat(entries.map((entry) => entry.name));
    } catch (err) {
        /* Filesystem unavailable: keep command-only completion. */
    }

    if (commandPosition && cwd !== '/') {
        try {
            const rootEntries = await listPath('/');
            candidates = candidates.concat(rootEntries.map((entry) => entry.name));
        } catch (err) {
            /* Ignore root completion failures. */
        }
    }

    candidates = Array.from(new Set(candidates.filter(Boolean)));

    const matches = partial
        ? candidates.filter((entry) => entry.toLowerCase().startsWith(partial.toLowerCase()))
        : candidates;

    return { matches, partial, commandPosition };
}

async function cmdTouch(arg) {
    if (!arg) {
        appendTextLine('Usage: touch <path> [more paths...]', 'terminal-muted');
        return;
    }

    const targets = String(arg).split(/\s+/).filter(Boolean);

    for (const target of targets) {
        const abs = resolvePath(target);

        if (abs === '/') {
            appendTextLine(
                formatError(createFsError('EINVAL', 'Cannot touch root directory', { path: abs })),
                'terminal-muted'
            );
            continue;
        }

        const existing = await tryStatPath(abs);

        if (existing) {
            if (isDirectory(existing)) {
                appendTextLine(
                    formatError(createFsError('EISDIR', 'Is a directory', { path: abs })),
                    'terminal-muted'
                );
                continue;
            }

            appendTextLine(`touched ${target} (existing)`, 'terminal-muted');
            continue;
        }

        const parent = parentPath(abs);
        const parentStat = await tryStatPath(parent);

        if (!parentStat) {
            throw createFsError('ENOENT', 'Parent directory not found', { path: abs, parent });
        }

        if (!isDirectory(parentStat)) {
            throw createFsError('ENOTDIR', 'Parent is not a directory', { path: abs, parent });
        }

        await writePath(abs, '');
        appendTextLine(`created ${target}`, 'terminal-muted');
    }
}

async function openVfsFile(absPath) {
    const st = await statPath(absPath);

    if (isDirectory(st)) {
        const indexAbs = joinPath(absPath, 'index.html');
        const indexStat = await tryStatPath(indexAbs);

        if (indexStat && isFile(indexStat)) {
            return openVfsFile(indexAbs);
        }

        if (isServableWebRoot()) {
            openWebpage(toWebUrl(absPath), pathBasename(absPath));
            return;
        }

        throw createFsError('ENOENT', 'No index.html found in directory', { path: absPath });
    }

    if (!isFile(st)) {
        throw createFsError('EINVAL', 'Unsupported file type', { path: absPath });
    }

    const name = pathBasename(absPath);
    const lower = String(absPath).toLowerCase();

    if (isServableWebRoot()) {
        if (lower.endsWith('.html') || lower.endsWith('.htm')) {
            openWebpage(toWebUrl(absPath), name);
        } else {
            window.open(toWebUrl(absPath), '_blank', 'noopener');
        }

        return;
    }

    const content = await readPath(absPath);
    const mime = lower.endsWith('.html') || lower.endsWith('.htm')
        ? 'text/html;charset=utf-8'
        : 'application/octet-stream';

    try {
        const blob = new Blob([String(content || '')], { type: mime });
        const url = URL.createObjectURL(blob);

        window.setTimeout(() => {
            try {
                URL.revokeObjectURL(url);
            } catch (err) {
                /* Ignore revoke failures. */
            }
        }, 5 * 60 * 1000);

        if (mime.startsWith('text/html')) {
            openWebpage(url, name);
        } else {
            window.open(url, '_blank', 'noopener');
        }
    } catch (err) {
        if (isServableWebRoot()) {
            if (lower.endsWith('.html') || lower.endsWith('.htm')) {
                openWebpage(toWebUrl(absPath), name);
            } else {
                window.open(toWebUrl(absPath), '_blank', 'noopener');
            }

            return;
        }

        throw normalizeFsError(err);
    }
}

                await removePath(abs);

        const prefix = target.endsWith('/') ? target : `${target}/`;
        const toRemove = saved.filter((k) => k.startsWith(prefix));

        if (toRemove.length) {
            toRemove.forEach((k) => removeSavedFile(k));
            appendTextLine(`Removed ${toRemove.length} saved files.`, 'terminal-muted');
            continue;
        }

        if (statError) {
            appendTextLine(formatError(statError), 'terminal-muted');
        } else {
            appendTextLine(
                formatError(createFsError('ENOENT', 'No such directory', { path: abs })),
                'terminal-muted'

        }
    }
}

async function cmdCp(input) {
    const parts = String(input).split(/\s+/);

    if (parts.length < 3) {
        appendTextLine('Usage: cp <source> <dest>', 'terminal-muted');
        return;
    }

    const source = parts[1];
    const dest = parts[2];
    const sourceAbs = resolvePath(source);
    let destAbs = resolvePath(dest);
    let content = null;
    let sourceError = null;

    try {
        const sourceStat = await statPath(sourceAbs);

        if (isDirectory(sourceStat)) {
            throw createFsError('EISDIR', 'Cannot copy directory', { path: sourceAbs });
        }

        content = await readPath(sourceAbs);
    } catch (err) {
        sourceError = normalizeFsError(err);

        if (!isMissingError(sourceError) && sourceError.code !== 'EBUSY') {
            throw sourceError;
        }

        let saved = loadFileFromStorage(source);

        if (saved === null) {
            saved = loadFileFromStorage(sourceAbs);
        }

        if (saved !== null) {
            content = saved;
        } else {
            throw sourceError;
        }
    }

    let destStat = null;

    try {
        destStat = await tryStatPath(destAbs);
    } catch (err) {
        const destError = normalizeFsError(err);

        if (destError.code !== 'EBUSY') {
            throw destError;
        }
    }

    if (destStat && isDirectory(destStat)) {
        destAbs = joinPath(destAbs, pathBasename(sourceAbs));
    }

    try {
        await writePath(destAbs, content);
        appendTextLine(`Copied ${source} -> ${destAbs}`, 'terminal-muted');
    } catch (err) {
        const writeError = normalizeFsError(err);

        if (writeError.code === 'EBUSY') {
            saveFileToStorage(destAbs, content);
            appendTextLine(
                `Filesystem unavailable; copied to browser storage: ${destAbs}`,
                'terminal-muted'
            );
        } else {
            throw writeError;
        }
    }
}

async function cmdSave(arg) {
    if (!currentEditor || !currentEditor.textarea) {
        appendTextLine('No open editor to save.', 'terminal-muted');
        return;
    }

    const requested = arg && arg.trim()
        ? arg.trim()
        : (currentEditor.path || currentEditor.filename || 'file.txt');

    const abs = resolvePath(requested);
    const value = currentEditor.textarea.value;

    try {
        await writePath(abs, value);
        currentEditor.path = abs;
        currentEditor.filename = pathBasename(abs);
        appendTextLine(`Saved to filesystem: ${abs}`, 'terminal-muted');
    } catch (err) {
        const writeError = normalizeFsError(err);

        if (writeError.code === 'EBUSY') {
            saveFileToStorage(requested, value);
            appendTextLine(
                `Filesystem unavailable; saved to browser storage: ${requested}`,
                'terminal-muted'
            );
        } else {
            appendTextLine(formatError(writeError), 'terminal-muted');
        }
    }
}

async function cmdLoad(arg) {
    if (!arg) {
        appendTextLine('Usage: load <name>', 'terminal-muted');
        return;
    }

    const name = arg.trim();
    const abs = resolvePath(name);
    let content = null;
    let sourcePath = null;

    try {
        const st = await tryStatPath(abs);

        if (st && isDirectory(st)) {
            throw createFsError('EISDIR', 'Is a directory', { path: abs });
        }

        if (st && isFile(st)) {
            content = await readPath(abs);
            sourcePath = abs;
        }
    } catch (err) {
        const loadError = normalizeFsError(err);

        if (!isMissingError(loadError) && loadError.code !== 'EBUSY') {
            throw loadError;
        }
    }

    if (content === null) {
        let saved = loadFileFromStorage(name);

        if (saved === null) {
            saved = loadFileFromStorage(abs);
        }

        if (saved !== null) {
            content = saved;
            sourcePath = name;
        }
    }

    if (content === null) {
        appendTextLine(`No saved file named ${name}`, 'terminal-muted');
        return;
    }

    openTextEditor(sourcePath || name, content, {
        editable: true,
        filename: pathBasename(sourcePath || name),
        path: sourcePath ? resolvePath(sourcePath) : abs
    });

    appendTextLine(`Loaded ${sourcePath || name} into editor`, 'terminal-muted');
}
